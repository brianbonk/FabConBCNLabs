# Python dependencies for the Fabric IQ workshop's Python tooling.
#
# ms-fabric-cli provides the `fab` command setup/provision_fabric_iq.py shells
# out to. pyyaml is used to parse setup/manifest.yaml. Neither script needs
# anything else -- both deliberately stick to the Python standard library
# (subprocess, argparse, sys, pathlib, re, csv, dataclasses) beyond these
# pins, per this workshop's "fool proof, minimal moving parts" design goal.
#
# Requires Python 3.10-3.13. ms-fabric-cli's latest PyPI release does not yet
# publish wheels for 3.14+ (`Requires-Python >=3.10,<3.14`), so this install
# fails outright on newer interpreters -- run setup/check_environment.py
# first, it checks this for you.
ms-fabric-cli>=1.0.0
pyyaml>=6.0

# No local Kusto dependency is needed to apply
# artifacts/Eventhouse/ColdChainKQLDB.kql -- provision_fabric_iq.py's
# run_kql_schema() runs it server-side via a throwaway Fabric notebook (`fab
# import` + `fab job run`, both reusing the existing `fab auth login`
# session), not from this laptop. An earlier version called Kusto directly
# via the azure-kusto-data SDK, which needed its own separate interactive
# sign-in; confirmed live that running from a notebook instead avoids that
# entirely (notebookutils.credentials.getToken("kusto") gives the notebook a
# trusted-execution token for free). See run_kql_schema()'s docstring.

# azure-eventhub is required ONLY for Module 02's lab, to run
# artifacts/generator/freezer_telemetry_generator.py (Fabric Eventstream's
# custom-endpoint source is Event Hubs-compatible SAS/AMQP, not a plain HTTPS
# endpoint -- see the generator script's docstring for why). It is not needed
# to run provision_fabric_iq.py itself.
azure-eventhub>=5.11.0
